// Copied into the pinned command package; no sandbox is applied to go test.
package worker

import (
	"reflect"
	"testing"

	"github.com/landlock-lsm/go-landlock/landlock"
	ll "github.com/landlock-lsm/go-landlock/landlock/syscall"
	"github.com/windtf/wireproxy"
)

func TestWarpPocketRoutingNetworkSandbox(t *testing.T) {
	info := "127.0.0.1:8090"
	bindOnly := landlock.MustConfig(landlock.AccessNetSet(ll.AccessNetBindTCP)).BestEffort()
	for _, test := range []struct {
		name     string
		sections []wireproxy.RoutineSpawner
		info     *string
		policy   landlock.Config
		rules    []landlock.Rule
	}{
		{"no SOCKS retains deny-by-default connect", nil, nil, landlock.V4.BestEffort(), nil},
		{
			"non-SOCKS retains existing connect and bind permissions",
			[]wireproxy.RoutineSpawner{&wireproxy.TCPServerTunnelConfig{Target: "203.0.113.1:8443"}},
			&info, landlock.V4.BestEffort(), []landlock.Rule{landlock.BindTCP(8090), landlock.ConnectTCP(8443)},
		},
		{
			"SOCKS direct capability retains exact listener restriction",
			[]wireproxy.RoutineSpawner{&wireproxy.Socks5Config{BindAddress: "127.0.0.1:25344"}},
			nil, bindOnly, []landlock.Rule{landlock.BindTCP(25344)},
		},
		{
			"mixed configuration retains every bind but no incompatible connect rule",
			[]wireproxy.RoutineSpawner{
				&wireproxy.TCPServerTunnelConfig{Target: "203.0.113.1:8443"},
				&wireproxy.Socks5Config{BindAddress: "127.0.0.1:25344"},
				&wireproxy.HTTPConfig{BindAddress: "127.0.0.1:8080"},
			},
			&info, bindOnly, []landlock.Rule{landlock.BindTCP(8090), landlock.BindTCP(25344), landlock.BindTCP(8080)},
		},
	} {
		t.Run(test.name, func(t *testing.T) {
			policy, rules := warpPocketNetworkPolicy(test.sections, test.info)
			if !reflect.DeepEqual(policy, test.policy) || !reflect.DeepEqual(rules, test.rules) {
				t.Fatalf("network policy = (%v, %v); want (%v, %v)", policy, rules, test.policy, test.rules)
			}
		})
	}
}
